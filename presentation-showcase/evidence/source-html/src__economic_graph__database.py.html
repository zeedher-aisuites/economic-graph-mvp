1"""Cliente MongoDB y comprobación de conectividad sin operaciones de escritura."""
2
3from typing import Any
4
5from pymongo import MongoClient
6from pymongo.errors import ConnectionFailure, OperationFailure, PyMongoError
7
8TIMEOUT_MS = 5_000
9
10
11def create_client(uri: str) -> MongoClient[dict[str, Any]]:
12 """Crear un cliente con tiempos acotados; no equivale a verificar la conexión."""
13 try:
14 return MongoClient(
15 uri,
16 serverSelectionTimeoutMS=TIMEOUT_MS,
17 connectTimeoutMS=TIMEOUT_MS,
18 socketTimeoutMS=TIMEOUT_MS,
19 timeoutMS=TIMEOUT_MS,
20 connect=False,
21 )
22 except (PyMongoError, ValueError):
23 # Los mensajes del driver pueden incluir la URI. No propagar esos detalles.
24 raise ConnectionError(
25 "No se pudo crear el cliente MongoDB. Revisa la URI y sus opciones."
26 ) from None
27
28
29def check_connection(uri: str, database_name: str) -> None:
30 """Ejecutar ping en la base indicada y cerrar el cliente incluso si falla."""
31 try:
32 with create_client(uri) as client:
33 response = client[database_name].command("ping")
34 if response.get("ok") != 1:
35 raise ConnectionError("MongoDB no confirmó correctamente la respuesta a ping.")
36 except OperationFailure as error:
37 if error.code in (13, 18):
38 raise ConnectionError(
39 "MongoDB rechazó la autenticación o los permisos. Revisa el usuario y sus accesos."
40 ) from None
41 raise ConnectionError(
42 "MongoDB rechazó ping. Revisa la configuración del servidor."
43 ) from None
44 except ConnectionFailure:
45 raise ConnectionError(
46 "No se pudo contactar con MongoDB dentro del tiempo límite. "
47 "Revisa el servidor, DNS, red y configuración TLS."
48 ) from None
49 except (PyMongoError, ValueError):
50 raise ConnectionError(
51 "No se pudo comprobar MongoDB. Revisa la URI, la base y la configuración TLS."
52 ) from None
