14def require_dedicated_database(database) -> None:
15 if database.name != MVP_DATABASE:
16 raise ValueError(f"Operación detenida: solo se permite la base dedicada {MVP_DATABASE}.")

19def _to_document(row: dict) -> dict:
20 result = deepcopy(row)
21 for field in ("observed_at", "recorded_at", "available_at"):
22 result[field] = parse_time(result[field])
23 result["_id"] = result["observation_id"]
24 return result

27def _from_document(document: dict) -> dict:
28 result = deepcopy(document)
29 result.pop("_id", None)
30 for field in ("observed_at", "recorded_at", "available_at"):
31 value = result[field]
32 # BSON conserva instantes UTC; PyMongo sin codec tz_aware los devuelve sin tzinfo.
33 if isinstance(value, datetime) and value.tzinfo is None:
34 value = value.replace(tzinfo=UTC)
35 result[field] = iso_time(value)
36 return result

39def load_observable(database, rows: list[dict]) -> dict:
40 """Agregar registros inmutables idempotentemente; jamás borrar ni reemplazar documentos."""
41 require_dedicated_database(database)
42 validate_observations(rows)
43 names = set(database.list_collection_names())
44 if names - {OBSERVATIONS, METADATA}:
45 raise ValueError("La base contiene colecciones ajenas; no se escribirá en ella.")
46 owner = database[METADATA].find_one({"_id": OWNER["_id"]})
47 if owner != OWNER and (owner is not None or OBSERVATIONS in names or METADATA in names):
48 raise ValueError("Las colecciones existentes no tienen una marca de propiedad compatible.")
49 collection = database[OBSERVATIONS]
50 # Revisar TODAS las colisiones antes de comenzar; no sobrescribir una versión distinta.
51 documents = [_to_document(row) for row in rows]
52 for document in documents:
53 existing = collection.find_one({"_id": document["_id"]})
54 if existing is not None and _from_document(existing) != _from_document(document):
55 raise ValueError(
56 f"{document['_id']}: existe otro contenido; carga detenida sin reemplazarlo."
57 )
58 database[METADATA].update_one({"_id": OWNER["_id"]}, {"$setOnInsert": OWNER}, upsert=True)
59 collection.create_index([("available_at", 1)], name="available_at")
60 collection.create_index([("subject_id", 1), ("available_at", 1)], name="subject_available")
61 inserted = 0
62 for document in documents:
63 result = collection.update_one(
64 {"_id": document["_id"]}, {"$setOnInsert": document}, upsert=True
65 )
66 inserted += result.upserted_id is not None
67 if _from_document(collection.find_one({"_id": document["_id"]})) != _from_document(
68 document
69 ):
70 raise ValueError("Colisión concurrente detectada; no se reemplazó ningún documento.")
71 return {
72 "observations_checked": len(rows),
73 "inserted": inserted,
74 "unchanged": len(rows) - inserted,
75 }

78def read_observations(database, as_of: str, company_id: str | None = None) -> list[dict]:
79 """Solo lectura sobre la colección pública; el filtro temporal se ejecuta en MongoDB."""
80 require_dedicated_database(database)
81 query = {"available_at": {"$lte": parse_time(as_of)}}
82 if company_id is not None:
83 if not isinstance(company_id, str) or not company_id.startswith("CO-"):
84 raise ValueError("La consulta de historial requiere una identidad observable CO-.")
85 query["subject_id"] = company_id
86 cursor = database[OBSERVATIONS].find(query)
87 rows = [_from_document(document) for document in cursor]
88 validate_observations(rows)
89 return rows
