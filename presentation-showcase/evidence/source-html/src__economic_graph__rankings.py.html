12POLICY = {
13 "version": "003a-conservative-v1",
14 "capacity": 5,
15 "additional_branches": {"weight": 10, "cap": 5},
16 "confirmed_neighbors": {"weight": 5, "cap": 4},
17 "baseline_rule": "10 × min(max(branch_count − 1, 0), 5)",
18 "graph_rule": "baseline + 5 × min(distinct_confirmed_economic_neighbors, 4)",
19 "eligibility_rule": "Conocidas salvo servicio activo confirmado sin conflicto.",
20 "tie_break": "Puntuación descendente, company_id público ascendente.",
21 "interpretation": "Puntos exploratorios; no probabilidades ni efecto económico medido.",
22}
23ACCEPTED_ATTRIBUTES = {"REPORTED", "CONFIRMED"}
24ECONOMIC_TYPES = {"SUPPLIES_TO", "PARTNERS_WITH"}

27def _latest(evidence: list[dict]) -> list[dict]:
28 instant = max(parse_time(row["available_at"]) for row in evidence)
29 return [row for row in evidence if parse_time(row["available_at"]) == instant]

32def _attribute_status(attribute: dict, evidence: list[dict]) -> str:
33 """No arbitrar valores distintos; tampoco usar una última evidencia descartada."""
34 if attribute["status"] != "OBSERVED":
35 return attribute["status"]
36 states = {row["verification_status"] for row in _latest(evidence)}
37 if len(states) > 1:
38 return "VERIFICATION_CONFLICTED"
39 return "OBSERVED" if states <= ACCEPTED_ATTRIBUTES else "INSUFFICIENT_EVIDENCE"

42def _branch_feature(attribute: dict, evidence: list[dict]) -> dict:
43 status = _attribute_status(attribute, evidence)
44 value = max(attribute["value"] - 1, 0) if status == "OBSERVED" else None
45 parameters = POLICY["additional_branches"]
46 return {
47 "status": status,
48 "observed_branch_count": attribute["value"],
49 "value": value,
50 **parameters,
51 "contribution": min(value, parameters["cap"]) * parameters["weight"]
52 if value is not None
53 else 0,
54 "rule": POLICY["baseline_rule"],
55 "evidence": evidence,
56 }

59def _relationship_context(relationships: list[dict]) -> list[dict]:
60 """Evaluar estados al corte; duplicados no multiplican vecinos ni ocultan conflictos."""
61 claims, by_signature = [], defaultdict(list)
62 for relationship in relationships:
63 evidence = relationship["status_history"]
64 values = {
65 (
66 row["value"]["relationship_type"],
67 row["value"]["from_company_id"],
68 row["value"]["to_company_id"],
69 )
70 for row in evidence
71 }
72 latest = _latest(evidence)
73 states = {row["verification_status"] for row in latest}
74 state = next(iter(states)) if len(states) == 1 and len(values) == 1 else "CONFLICTED"
75 claim = {
76 "relationship_id": relationship["relationship_id"],
77 **latest[0]["value"],
78 "verification_status": state,
79 "scoring_status": state,
80 "evidence": evidence,
81 }
82 if claim["relationship_type"] not in ECONOMIC_TYPES:
83 claim["scoring_status"] = "NOT_ECONOMIC"
84 claims.append(claim)
85 # Conservar IDs e historia; agrupar hechos equivalentes solo para detectar oposición.
86 for signature in values:
87 by_signature[signature].append(claim)
88 for group in by_signature.values():
89 states = {claim["verification_status"] for claim in group}
90 if "CONFIRMED" in states and states & {"DISCARDED", "CONFLICTED"}:
91 for claim in group:
92 if claim["relationship_type"] in ECONOMIC_TYPES:
93 claim["scoring_status"] = "CONFLICTED"
94 return claims

97def _neighbor_feature(company_id: str, claims: list[dict]) -> dict:
98 related = [
99 claim
100 for claim in claims
101 if any(
102 company_id in (row["value"]["from_company_id"], row["value"]["to_company_id"])
103 for row in claim["evidence"]
104 )
105 ]
106 neighbors = defaultdict(list)
107 for claim in related:
108 if claim["scoring_status"] == "CONFIRMED":
109 other = (
110 claim["to_company_id"]
111 if claim["from_company_id"] == company_id
112 else (claim["from_company_id"])
113 )
114 neighbors[other].append(claim["relationship_id"])
115 value = len(neighbors)
116 parameters = POLICY["confirmed_neighbors"]
117 return {
118 "status": "OBSERVED" if value else "INSUFFICIENT_EVIDENCE",
119 "value": value,
120 **parameters,
121 "contribution": min(value, parameters["cap"]) * parameters["weight"],
122 "rule": "Vecinos distintos en SUPPLIES_TO/PARTNERS_WITH CONFIRMED; sin fusión de alias.",
123 "neighbors": [
124 {"company_id": key, "relationship_ids": sorted(neighbors[key])}
125 for key in sorted(neighbors)
126 ],
127 "relationship_ids": [claim["relationship_id"] for claim in related],
128 "evidence": [row for claim in related for row in claim["evidence"]],
129 }

132def _boundary(ordered: list[dict], score_key: str) -> dict:
133 capacity = POLICY["capacity"]
134 if not ordered:
135 return {"crosses_tie": False, "score": None, "tied_company_ids": []}
136 index = min(capacity, len(ordered)) - 1
137 score = ordered[index][score_key]
138 return {
139 "crosses_tie": len(ordered) > capacity and ordered[capacity][score_key] == score,
140 "score": score,
141 "tied_company_ids": [row["company_id"] for row in ordered if row[score_key] == score],
142 }

145def compare_rankings(rows: list[dict], as_of: str | datetime) -> dict:
146 """Calcular ambas políticas sobre la misma cohorte, sin red, escrituras ni etiquetas."""
147 view = snapshot(rows, as_of)
148 observations = view["observations"]
149 claims = _relationship_context(view["relationships"])
150 companies, excluded = [], []
151 for company in view["companies"]:
152 company_id, attributes = company["company_id"], company["attributes"]
153 evidence = defaultdict(list)
154 for row in observations:
155 if row["subject_type"] == "COMPANY" and row["subject_id"] == company_id:
156 evidence[row["attribute"]].append(row)
157 service = attributes["service_active"]
158 service_evidence = evidence["service_active"]
159 service_status = _attribute_status(service, service_evidence)
160 active = (
161 service_status == "OBSERVED"
162 and service["value"] is True
163 and {row["verification_status"] for row in _latest(service_evidence)} == {"CONFIRMED"}
164 )
165 result = {
166 "company_id": company_id,
167 "display_name": attributes["display_name"]["value"],
168 "name_evidence": evidence["display_name"],
169 "identity_status": company["identity_status"],
170 "eligibility": {
171 "eligible": not active,
172 "status": "ACTIVE_SERVICE_CONFIRMED" if active else "PENDING_COMMERCIAL_VALIDATION",
173 "service_status": service_status,
174 "observed_service_active": service["value"],
175 "evidence": service_evidence,
176 },
177 }
178 if active:
179 excluded.append(result)
180 continue
181 branches = _branch_feature(attributes["branch_count"], evidence["branch_count"])
182 neighbors = _neighbor_feature(company_id, claims)
183 result.update(
184 {
185 "features": {"additional_branches": branches, "confirmed_neighbors": neighbors},
186 "baseline_score": branches["contribution"],
187 "graph_score": branches["contribution"] + neighbors["contribution"],
188 }
189 )
190 companies.append(result)
191 baseline = sorted(companies, key=lambda row: (-row["baseline_score"], row["company_id"]))
192 graph = sorted(companies, key=lambda row: (-row["graph_score"], row["company_id"]))
193 capacity = POLICY["capacity"]
194 for policy, ordered in (("baseline", baseline), ("graph", graph)):
195 for position, company in enumerate(ordered, start=1):
196 company[f"{policy}_position"] = position
197 company[f"{policy}_selected"] = position <= capacity
198 for company in baseline:
199 company["position_change"] = company["baseline_position"] - company["graph_position"]
200 company["selection_change"] = (
201 "BOTH"
202 if company["baseline_selected"] and company["graph_selected"]
203 else "GRAPH_ONLY"
204 if company["graph_selected"]
205 else "BASELINE_ONLY"
206 if company["baseline_selected"]
207 else "NEITHER"
208 )
209 baseline_ids = [row["company_id"] for row in baseline[:capacity]]
210 graph_ids = [row["company_id"] for row in graph[:capacity]]
211 canonical = json.dumps(observations, sort_keys=True, ensure_ascii=False, separators=(",", ":"))
212 return {
213 "policy": deepcopy(POLICY),
214 "as_of": view["as_of"],
215 "visible_input_sha256": hashlib.sha256(canonical.encode("utf-8")).hexdigest(),
216 "known_companies": len(view["companies"]),
217 "eligible_companies": len(companies),
218 "visible_observations": len(observations),
219 "companies": baseline,
220 "excluded": excluded,
221 "top_k": {"baseline": baseline_ids, "graph": graph_ids},
222 "comparison": {
223 "shared": sorted(set(baseline_ids) & set(graph_ids)),
224 "entered_graph": [key for key in graph_ids if key not in baseline_ids],
225 "left_baseline": [key for key in baseline_ids if key not in graph_ids],
226 "baseline_boundary": _boundary(baseline, "baseline_score"),
227 "graph_boundary": _boundary(graph, "graph_score"),
228 "unfilled_capacity": max(capacity - len(companies), 0),
229 },
230 "relationship_context": claims,
231 "identity_candidates": view["identities"],
232 "conflicts": view["conflicts"],
233 }

239 def label(value: object) -> str:
240 return " ".join(str(value).split()).replace("|", "\\|")

242 names = {
243 row["company_id"]: row["display_name"] or row["company_id"] for row in report["companies"]

287 [

294 lines.extend(
295 [
296 "",
297 "Excluidas por servicio activo confirmado: "
