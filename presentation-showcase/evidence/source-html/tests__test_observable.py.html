201def test_public_contract_rejects_invalid_records(projection, mutation):
202 rows = projection[0]
203 relation = next(row for row in rows if row["subject_type"] == "RELATIONSHIP")
204 if mutation == "private_nested":
205 relation["value"]["metadata"] = {"truth_exists": True}
206 elif mutation == "duplicate":
207 rows.append(deepcopy(rows[0]))
208 elif mutation == "orphan":

210 elif mutation == "self":
211 relation["value"]["from_company_id"] = relation["value"]["to_company_id"]
212 elif mutation == "unknown_type":
213 relation["value"]["relationship_type"] = "SAME_BUSINESS_AS"
214 elif mutation == "source":
215 rows[0]["source_type"] = "UNKNOWN"
216 elif mutation == "date":
217 rows[0]["available_at"] = "ayer"
218 elif mutation == "time_order":
219 rows[0]["recorded_at"] = "2027-01-01T12:00:00Z"
