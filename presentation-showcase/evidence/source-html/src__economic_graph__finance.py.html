25def _decimal(value: object, label: str) -> Decimal:
26 # Exigir cadenas evita introducir redondeo binario desde números JSON float.
27 if not isinstance(value, str):
28 raise ValueError(f"{label}: usa una cadena decimal, por ejemplo '0.02', sin símbolo %.")
29 try:
30 result = Decimal(value)
31 if not result.is_finite():
32 raise InvalidOperation
33 except InvalidOperation:
34 raise ValueError(f"{label}: decimal finito inválido.") from None
35 return result

69def _money(spec: dict, unit: str, settings: dict) -> Decimal:
70 if not isinstance(spec, dict) or spec.get("unit") != unit:
71 raise ValueError(f"Importe: se requiere unidad {unit}.")
72 if spec.get("basis") == "present_value":
73 _fields(spec, {"unit", "basis", "value", "horizon_months", "monthly_discount_rate"}, unit)
74 if (
75 type(spec["horizon_months"]) is not int
76 or spec["horizon_months"] != settings["horizon_months"]
77 or _decimal(spec["monthly_discount_rate"], unit) != settings["monthly_discount_rate"]
78 ):
79 raise ValueError(
80 "Valor presente incompatible con horizonte/descuento; revalora los importes."
81 )
82 # Ya descontado: volver a descontar aquí duplicaría el efecto financiero.
83 return _decimal(spec["value"], unit)
84 if spec.get("basis") == "monthly_cashflows":
85 _fields(spec, {"unit", "basis", "values"}, unit)
86 values = spec["values"]
87 if not isinstance(values, list) or len(values) != settings["horizon_months"] + 1:
88 raise ValueError("Flujos mensuales: incluye exactamente mes 0 hasta H, sin truncarlos.")
89 return sum(
90 (
91 _decimal(value, unit) / (1 + settings["monthly_discount_rate"]) ** month
92 for month, value in enumerate(values)
93 ),
94 Decimal(0),
95 )
96 raise ValueError("Importe: basis debe ser present_value o monthly_cashflows.")

99def _scenario(scenario: dict, settings: dict) -> dict:
100 money_units = {
101 "contract_margin_per_activation": "MXN_per_activation",
102 "baseline_commercial_cost_per_slot": "MXN_per_slot",
103 "incremental_commercial_cost_per_slot": "MXN_per_slot",
104 "unselected_margin_increment": "MXN_total",
105 }
106 _fields(
107 scenario,
108 set(money_units)
109 | {
110 "name",
111 "activation_window_days",
112 "baseline_activation_rate",
113 "activation_rate_increment",
114 "contract_margin_scope",
115 "commercial_cost_scope",
116 "costs",
117 },
118 "scenario",
119 )
120 if scenario["name"] not in (*SCENARIOS, "null"):
121 raise ValueError("Escenario desconocido.")
122 if (
123 type(scenario["activation_window_days"]) is not int
124 or scenario["activation_window_days"] != settings["activation_window_days"]
125 ):
126 raise ValueError("La ventana de las tasas/márgenes no coincide con settings.")
127 if (
128 scenario["contract_margin_scope"] != "net_of_service_installation_commission"
129 or scenario["commercial_cost_scope"] != "prospecting_only"
130 ):
131 raise ValueError("Separar margen neto de servicio/alta/comisión y costo de prospección.")
132 result = {key: _money(scenario[key], unit, settings) for key, unit in money_units.items()}
133 result["name"] = scenario["name"]
134 p = _decimal(scenario["baseline_activation_rate"], "baseline_activation_rate")
135 delta = _decimal(scenario["activation_rate_increment"], "activation_rate_increment")
136 if not 0 <= p <= 1 or not 0 <= p + delta <= 1:
137 raise ValueError("Las probabilidades baseline y grafo deben estar entre 0 y 1.")
138 result.update(baseline_activation_rate=p, activation_rate_increment=delta)
139 if scenario["name"] == "null" and (delta != 0 or result["unselected_margin_increment"] != 0):
140 raise ValueError("El escenario nulo requiere Δp=0 y ΔU=0.")
141 if (
142 result["baseline_commercial_cost_per_slot"] < 0
143 or result["baseline_commercial_cost_per_slot"]
144 + result["incremental_commercial_cost_per_slot"]
145 < 0
146 ):
147 raise ValueError("El costo comercial total por plaza no puede ser negativo.")
148 result.update(fixed_incremental_cost=Decimal(0), operating_incremental_cost=Decimal(0))
149 seen, categories = set(), set()
150 if not isinstance(scenario["costs"], list):
151 raise ValueError("costs debe ser una lista con costos fijos y operativos explícitos.")
152 for cost in scenario["costs"]:
153 _fields(cost, {"cost_id", "category", "amount"}, "cost")
154 identifier, category = cost["cost_id"], cost["category"]
155 if not isinstance(identifier, str) or not identifier.strip() or identifier in seen:
156 raise ValueError("cost_id vacío o duplicado: evitar doble contabilización.")
157 if category not in ("fixed", "operating"):
158 raise ValueError(
159 "Solo costos incrementales fixed/operating; prospección se cuenta aparte."
160 )
161 amount = _money(cost["amount"], "MXN_total", settings)
162 if amount < 0:
163 raise ValueError(
164 "Los costos fijos y operativos de esta versión deben ser no negativos."
165 )
166 seen.add(identifier)
167 categories.add(category)
168 result[f"{category}_incremental_cost"] += amount
169 if categories != {"fixed", "operating"}:
170 raise ValueError("Declarar costos fixed y operating, incluso cuando sean cero.")
171 return result

174def _threshold(cost: Decimal, slots: int, margin: Decimal, p: Decimal) -> dict:
175 if margin <= 0:
176 return {"delta_rate": None, "delta_pp": None, "feasibility": "NON_POSITIVE_MARGIN"}
177 rate = cost / (slots * margin)
178 feasibility = "ATTAINABLE"
179 if rate > 1 - p:
180 feasibility = "UNATTAINABLE"
181 elif rate < -p:
182 feasibility = "BELOW_MINIMUM_RATE"
183 return {"delta_rate": rate, "delta_pp": rate * 100, "feasibility": feasibility}

186def _evaluate(scenario: dict, settings: dict, pools: int) -> dict:
187 pools = _positive_integer(pools, "pools")
188 slots = pools * settings["capacity_per_pool"]
189 p, delta = scenario["baseline_activation_rate"], scenario["activation_rate_increment"]
190 margin = scenario["contract_margin_per_activation"]
191 commercial_base = slots * scenario["baseline_commercial_cost_per_slot"]
192 commercial_graph = commercial_base + slots * scenario["incremental_commercial_cost_per_slot"]
193 fixed, operating = scenario["fixed_incremental_cost"], scenario["operating_incremental_cost"]
194 commercial_delta = commercial_graph - commercial_base
195 unselected = scenario["unselected_margin_increment"]
196 selected_delta = slots * delta * margin
197 net = selected_delta - commercial_delta + unselected - fixed - operating
198 reserve = settings["reserve_rate"] * (fixed + operating)
199 threshold_cost = commercial_delta + fixed + operating - unselected
200 return {
201 "scenario": scenario["name"],
202 "pools": pools,
203 "capacity_per_pool": 5,
204 "slots": slots,
205 "baseline_activation_rate": p,
206 "graph_activation_rate": p + delta,
207 "activation_rate_increment": delta,
208 "contract_margin_per_activation_mxn": margin,
209 "baseline_selected_margin_mxn": slots * p * margin,
210 "graph_selected_margin_mxn": slots * (p + delta) * margin,
211 "selected_margin_increment_mxn": selected_delta,
212 "baseline_commercial_cost_mxn": commercial_base,
213 "graph_commercial_cost_mxn": commercial_graph,
214 "commercial_cost_increment_mxn": commercial_delta,
215 "unselected_margin_increment_mxn": unselected,
216 "fixed_incremental_cost_mxn": fixed,
217 "operating_incremental_cost_mxn": operating,
218 "net_present_value_mxn": net,
219 "balance_status": "POSITIVE" if net > 0 else "NEGATIVE" if net < 0 else "ZERO",
220 "reserve_mxn": reserve,
221 "net_after_reserve_mxn": net - reserve,
222 "clears_reserve_mathematically": net > reserve,
223 "break_even": _threshold(threshold_cost, slots, margin, p),
224 "required_with_reserve": _threshold(threshold_cost + reserve, slots, margin, p),
225 "maximum_fixed_and_operating_budget_mxn": selected_delta - commercial_delta + unselected,
226 "operational_authorization": False,
227 }

230def _plain(value):
231 if isinstance(value, Decimal):
232 if value == 0:
233 return "0"
234 text = format(value, "f")
235 return text.rstrip("0").rstrip(".") if "." in text else text
236 if isinstance(value, dict):
237 return {key: _plain(item) for key, item in value.items()}
238 if isinstance(value, list):
239 return [_plain(item) for item in value]
240 return value

269def _sensitivity(base: dict, settings: dict, pools: int) -> dict:
270 ranges = {
271 "activation_rate_increment": ["-0.02", "0", "0.01", "0.02", "0.05"],
272 "contract_margin_per_activation": ["4000", "8000", "12000"],
273 "incremental_commercial_cost_per_slot": ["25", "50", "100"],
274 "fixed_incremental_cost": ["5000", "10000", "20000"],
275 "operating_incremental_cost": ["3000", "6000", "12000"],
276 "unselected_margin_increment": ["-1000", "0", "1000"],
277 "reserve_rate": ["0", "0.20", "0.50"],
278 "pools": [1, 10, 100],
279 }
280 one_way, rate_margin, volume_cost = [], [], []
281 for parameter, values in ranges.items():
282 for value in values:
283 case, case_settings, volume = deepcopy(base), deepcopy(settings), pools
284 if parameter == "pools":
285 volume = value
286 elif parameter == "reserve_rate":
287 case_settings[parameter] = Decimal(value)
288 else:
289 case[parameter] = Decimal(value)
290 # Conservar como inválidas las variaciones que excedan probabilidades posibles.
291 p = case["baseline_activation_rate"] + case["activation_rate_increment"]
292 if not 0 <= p <= 1:
293 one_way.append({"parameter": parameter, "value": value, "status": "INVALID_RATE"})
294 continue
295 one_way.append(
296 {
297 "parameter": parameter,
298 "value": value,
299 "result": _evaluate(case, case_settings, volume),
300 }
301 )
302 for rate in ranges["activation_rate_increment"]:
303 for margin in ranges["contract_margin_per_activation"]:
304 case = {
305 **base,
306 "activation_rate_increment": Decimal(rate),
307 "contract_margin_per_activation": Decimal(margin),
308 }
309 if 0 <= case["baseline_activation_rate"] + case["activation_rate_increment"] <= 1:
310 rate_margin.append(_evaluate(case, settings, pools))
311 else:
312 rate_margin.append(
313 {
314 "activation_rate_increment": rate,
315 "contract_margin_per_activation_mxn": margin,
316 "status": "INVALID_RATE",
317 }
318 )
319 for volume in ranges["pools"]:
320 for factor in (Decimal("0.5"), Decimal(1), Decimal(2)):
321 case = {
322 **base,
323 "fixed_incremental_cost": base["fixed_incremental_cost"] * factor,
324 "operating_incremental_cost": base["operating_incremental_cost"] * factor,
325 }
326 volume_cost.append({"cost_factor": factor, "result": _evaluate(case, settings, volume)})
327 return {
328 "one_way": one_way,
329 "rate_by_margin": rate_margin,
330 "volume_by_cost": volume_cost,
331 "warning": "Rangos exploratorios, no intervalos de confianza. Al variar volumen, costos "
332 "constantes solo aíslan su efecto; no son cotizaciones a escala. H y descuento "
333 "requieren revalorar los importes o aportar flujos mensuales compatibles.",
334 }

337def financial_report(config: dict, scope: str = "both", sensitivity: bool = False) -> dict:
338 """Tres escenarios y nulo obligatorio; cinco y 500 plazas separados del experimento."""
339 _fields(config, {"schema_version", "assumption_status", "settings", "scenarios"}, "config")
340 if config["schema_version"] != VERSION or config["assumption_status"] != STATUS:
341 raise ValueError("Contrato financiero o etiqueta educativa no calibrada inválidos.")
342 if scope not in {"demo", "hypothetical", "both"}:
343 raise ValueError("scope debe ser demo, hypothetical o both.")
344 with localcontext() as context:
345 context.prec, context.rounding = 50, ROUND_HALF_EVEN
346 settings = _settings(config["settings"])
347 if not isinstance(config["scenarios"], list):
348 raise ValueError("scenarios debe ser una lista.")
349 cases = [_scenario(item, settings) for item in config["scenarios"]]
350 if len(cases) != 3 or {case["name"] for case in cases} != set(SCENARIOS):
351 raise ValueError("Incluir conservative, base y optimistic exactamente una vez.")
352 cases = sorted(cases, key=lambda case: SCENARIOS.index(case["name"]))
353 base = cases[1]
354 cases.append(
355 {
356 **base,
357 "name": "null",
358 "activation_rate_increment": Decimal(0),
359 "unselected_margin_increment": Decimal(0),
360 }
361 )
362 volumes = {"demo": 1, "hypothetical": 100}
363 results = []
364 analyses = {}
365 for name, pools in volumes.items():
366 if scope != "both" and scope != name:
367 continue
368 results.extend({"scope": name, **_evaluate(case, settings, pools)} for case in cases)
369 if sensitivity:
370 analyses[name] = _sensitivity(base, settings, pools)
371 canonical = json.dumps(config, ensure_ascii=False, sort_keys=True, separators=(",", ":"))
372 return _plain(
373 {
374 "schema_version": VERSION,
375 "assumption_status": STATUS,
376 "warning": WARNING,
377 "settings": settings,
378 "input_sha256": hashlib.sha256(canonical.encode()).hexdigest(),
379 "results": results,
380 "sensitivity": analyses,
381 "zero_balance_reference": [
382 {
383 "scope": row["scope"],
384 "delta_rate": row["break_even"]["delta_rate"],
385 "feasibility": row["break_even"]["feasibility"],
386 "net_present_value_mxn": "0",
387 "definition": "Identidad algebraica ΔV=0; "
388 "no es un efecto observado. Una tasa decimal periódica es aproximada.",
389 }
390 for row in results
391 if row["scenario"] == "base" and row["break_even"]["delta_rate"] is not None
392 ],
393 "future_experiment": {
394 "status": "future_methodological_design_only",
395 "randomization": "1:1 por bolsas independientes",
396 "capacity_per_pool": 5,
397 "confidence_level": "0.95",
398 "target_power": "0.80",
399 "sample_size": None,
400 "inference": "Inviable sin suficientes bolsas independientes; faltan varianza, "
401 "correlación intrabolsa y datos comerciales.",
402 "operational_authorization": False,
