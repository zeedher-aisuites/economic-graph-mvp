202def install_read_only_client(monkeypatch, rows):
203 """No ofrece ping, metadata, carga, índices ni métodos de escritura."""
204 calls = []
205
206 class Collection:
207 def find(self, query):
208 calls.append(query)
209 return [
210 {**r, "_id": r["observation_id"]}
211 for r in rows
212 if parse_time(r["available_at"]) <= query["available_at"]["$lte"]
213 ]
214
215 class Database:
216 name = MVP_DATABASE
217
218 def __getitem__(self, name):
219 assert name == OBSERVATIONS
220 return Collection()
221
222 class Client:
223 def __enter__(self):
224 calls.append("open")
225 return self
226
227 def __exit__(self, *args):
228 calls.append("closed")
229
230 def __getitem__(self, name):
231 assert name == MVP_DATABASE
232 return Database()
233
234 monkeypatch.setattr(tower, "create_client", lambda uri: Client())
235 monkeypatch.setattr(tower, "load_config", lambda path: ("mongodb://fake", MVP_DATABASE))
236 return calls
