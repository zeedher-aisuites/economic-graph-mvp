22def public_id(prefix: str, observation_id: str) -> str:
23 """Asignar identidad estable desde evidencia, independiente del ordinal EMP o CLM."""
24 return f"{prefix}-{uuid5(NAMESPACE_URL, 'conectapyme/1.0/' + observation_id).hex}"

27def project_observations(plan: list[dict]) -> tuple[list[dict], dict]:
28 """Devolver ledger público y conciliación privada; ninguna decisión usa la verdad."""
29 company_map = {
30 row["subject_id"]: public_id("CO", row["observation_id"])
31 for row in plan
32 if row["subject_type"] == "COMPANY" and row["attribute"] == "display_name"
33 }
34 subjects = {}
35 for row in sorted(
36 plan, key=lambda item: (iso_time(item["available_at"]), item["observation_id"])
37 ):
38 kind = row["subject_type"]
39 prefix = "REC" if kind == "COMPANY_ALIAS" else "LINK"
40 if kind != "COMPANY":
41 subjects.setdefault(row["subject_id"], public_id(prefix, row["observation_id"]))
42 result = []
43 for original in sorted(plan, key=lambda item: item["observation_id"]):
44 identifier = original["observation_id"]
45 kind = original["subject_type"]
46 row = {
47 key: deepcopy(original[key])
48 for key in (
49 "observation_id",
50 "attribute",
51 "value",
52 "source_type",
53 "verification_status",
54 "observed_at",
55 "recorded_at",
56 "available_at",
57 "evidence_note",
58 "conflict_group",
59 )
60 }
61 row["source_record_id"] = f"SRC-{original['source_type']}-{identifier}"
62 row["subject_type"] = {"REFERRAL": "RELATIONSHIP", "COMPANY_ALIAS": "SOURCE_RECORD"}.get(
63 kind, kind
64 )
65 row["subject_id"] = (
66 company_map[original["subject_id"]]
67 if kind == "COMPANY"
68 else subjects[original["subject_id"]]
69 )
70 if kind in {"RELATIONSHIP", "REFERRAL"}:
71 value = original["value"]
72 left_key, right_key = (
73 ("referrer_company_id", "referred_company_id")
74 if kind == "REFERRAL"
75 else ("from_company_id", "to_company_id")
76 )
77 try:
78 left, right = company_map[value[left_key]], company_map[value[right_key]]
79 except KeyError:
80 raise ValueError(
81 f"{identifier}: extremo sin descubrimiento; proyección detenida."
82 ) from None
83 relationship_type = "REFERS_TO" if kind == "REFERRAL" else value["relationship_type"]
84 if relationship_type == "PARTNERS_WITH":
85 left, right = sorted((left, right))
86 row["attribute"] = "relationship_claim"
87 row["value"] = {
88 "from_company_id": left,
89 "to_company_id": right,
90 "relationship_type": relationship_type,
91 }
92 row["evidence_note"] = EVIDENCE_PROJECTION.get(identifier, row["evidence_note"])
93 for field in ("observed_at", "recorded_at", "available_at"):
94 row[field] = iso_time(row[field])
95 result.append(row)
96 validate_observations(result)
97 return result, {"companies": company_map, "subjects": subjects}

100def write_projection(
101 rows: list[dict], reconciliation: dict, output: Path = Path("data/observable")
102) -> None:
103 """Escribir únicamente los dos artefactos propios; no borrar archivos del usuario."""
104 validate_observations(rows)
105 output.mkdir(parents=True, exist_ok=True)
106 content = (
107 json.dumps({"schema_version": "1.0", "observations": rows}, ensure_ascii=False, indent=2)
108 + "\n"
109 )
110 target = output / "observations.json"
111 if not target.exists() or target.read_text(encoding="utf-8") != content:
112 target.write_text(content, encoding="utf-8")
113 private = Path("data/synthetic/restricted")
