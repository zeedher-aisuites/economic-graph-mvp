20class MemoryCollection:
21 """Doble inmutable: solo implementa las operaciones permitidas por el cargador."""
22
23 def __init__(self):
24 self.documents = {}
25 self.indexes = {}
26 self.created = False
27 self.writes = 0
28 self.queries = []
29
30 def find_one(self, query):
31 return deepcopy(self.documents.get(query["_id"]))
32
33 def update_one(self, query, update, upsert):
34 assert upsert and set(update) == {"$setOnInsert"}
35 self.created = True
36 key = query["_id"]
37 inserted = key not in self.documents
38 if inserted:
39 self.documents[key] = deepcopy(update["$setOnInsert"])
40 self.writes += 1
41 return SimpleNamespace(upserted_id=key if inserted else None)
42
43 def create_index(self, keys, name):
44 self.created = True
45 self.indexes[name] = keys
46
47 def find(self, query):
48 self.queries.append(deepcopy(query))
49 cut = query["available_at"]["$lte"]
50 return [
51 deepcopy(doc)
52 for doc in self.documents.values()
53 if parse_time(doc["available_at"]) <= cut
54 and ("subject_id" not in query or doc["subject_id"] == query["subject_id"])
55 ]

70def test_repeated_load_is_idempotent_for_observations_companies_and_edges(projection):
71 rows, _ = projection
72 database = MemoryDatabase()
73 assert load_observable(database, rows) == {
74 "observations_checked": 83,
75 "inserted": 83,
76 "unchanged": 0,
77 }
78 before = deepcopy(database[OBSERVATIONS].documents)
79 assert load_observable(database, rows) == {
80 "observations_checked": 83,
81 "inserted": 0,
82 "unchanged": 83,
83 }
84 assert database[OBSERVATIONS].documents == before
85 assert len(before) == 83
86 for cut, companies, claims in [("T0", 12, 2), ("T1", 17, 13), ("T2", 23, 26)]:
87 summary = summarize(snapshot(read_observations(database, cut), cut))
88 assert summary["companies"] == companies
89 assert summary["relationship_claims"] == claims
90 assert database[OBSERVATIONS].queries[-1] == {"available_at": {"$lte": parse_time("T2")}}

133def test_naive_bson_utc_dates_are_normalized_on_read(projection):
134 database = MemoryDatabase()
135 load_observable(database, projection[0])
136 for doc in database[OBSERVATIONS].documents.values():
137 for field in ("observed_at", "recorded_at", "available_at"):
138 doc[field] = doc[field].replace(tzinfo=None)
139 # El doble emula la comparación de instantes BSON del servidor.
140 collection = database[OBSERVATIONS]
141 collection.find = lambda query: [
142 deepcopy(doc)
143 for doc in collection.documents.values()
144 if doc["available_at"] <= query["available_at"]["$lte"].replace(tzinfo=None)
145 ]
146 assert len(snapshot(read_observations(database, "T0"), "T0")["companies"]) == 12
