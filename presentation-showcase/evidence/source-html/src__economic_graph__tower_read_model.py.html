34def sector(company: dict) -> str:
35 attribute = company["attributes"]["sector"]
36 if attribute["status"] == "CONFLICTED":
37 return CONFLICTED
38 return attribute["value"] if attribute["status"] == "OBSERVED" else UNKNOWN

41def build_context(rows: list[dict], cut: str) -> dict:
42 """Calcular siempre sobre la entrada completa al corte, antes de filtros visuales."""
43 view = snapshot(rows, cut)
44 ranking = compare_rankings(rows, cut)
45 eligible = ranking["companies"]
46 return {
47 "view": view,
48 "summary": summarize(view),
49 "ranking": ranking,
50 "evolution": [
51 {"Corte": label, **summarize(snapshot(rows, instant))}
52 for label, instant in CUTS.items()
53 if parse_time(instant) <= parse_time(cut)
54 ],
55 "sources": dict(Counter(row["source_type"] for row in view["observations"])),
56 "states": dict(Counter(row["verification_status"] for row in view["relationships"])),
57 "coverage": {
58 "branches": sum(
59 row["features"]["additional_branches"]["status"] == "OBSERVED" for row in eligible
60 ),
61 "neighbors": sum(
62 row["features"]["confirmed_neighbors"]["value"] > 0 for row in eligible
63 ),
64 "denominator": len(eligible),
65 },
66 }

99def funnel(context: dict, ids: set, policy: str) -> list[dict]:
100 report = context["ranking"]
101 eligible = {row["company_id"] for row in report["companies"]}
102 excluded = {row["company_id"] for row in report["excluded"]}
103 measured = [
104 ("Conocidas", len(ids)),
105 ("Elegibles para ranking", len(ids & eligible)),
106 ("Excluidas (rama aparte)", len(ids & excluded)),
107 ("Priorizadas del top-5 global", len(ids & set(report["top_k"][policy]))),
108 ]
109 return [
110 {"Etapa": name, "Cantidad": count, "Estado": "Sintético observable"}
111 for name, count in measured
112 ] + [
113 {"Etapa": name, "Cantidad": None, "Estado": "No observado"}
114 for name in ("Contactos", "Contactos activados", "Oportunidades", "Activaciones", "Ventas")
115 ]

118def account_detail(context: dict, company_id: str) -> dict | None:
119 view, report = context["view"], context["ranking"]
120 company = next((c for c in view["companies"] if c["company_id"] == company_id), None)
121 if company is None:
122 return None
123 ranking = next(
124 c for c in report["companies"] + report["excluded"] if c["company_id"] == company_id
125 )
126 relationships = [
127 claim
128 for claim in report["relationship_context"]
129 if any(
130 company_id in (row["value"]["from_company_id"], row["value"]["to_company_id"])
131 for row in claim["evidence"]
132 )
133 ]
134 candidates = [row for row in view["identities"] if company_id in row["candidate_company_ids"]]
135 subjects = (
136 {company_id}
137 | {row["relationship_id"] for row in relationships}
138 | {row["source_record_id"] for row in candidates}
139 )
140 return {
141 "company": company,
142 "ranking": ranking,
143 "relationships": relationships,
144 "candidates": candidates,
145 "observations": [row for row in view["observations"] if row["subject_id"] in subjects],
146 }

149def network_data(context: dict, ids: set, focus_id: str | None = None) -> dict:
150 """Red incidente al segmento; extremos externos conocidos permanecen como contexto."""
151 view, report = context["view"], context["ranking"]
152 known = {row["company_id"]: row for row in view["companies"]}
153 if not ids <= known.keys():
154 raise ValueError("La selección contiene una empresa no conocida al corte.")
155 if focus_id is not None and ids != {focus_id}:
156 raise ValueError("El foco debe corresponder a la única empresa seleccionada.")
157 claims = [
158 row for row in view["relationships"] if ids & {row["from_company_id"], row["to_company_id"]}
159 ]
160 diagnostics = {row["relationship_id"]: row for row in report["relationship_context"]}
161 shown = ids | {row[key] for row in claims for key in ("from_company_id", "to_company_id")}
162 if not shown <= known.keys():
163 raise ValueError("La red contiene un extremo no identificable al corte.")
164 ordered = sorted(shown - {focus_id}) if focus_id else sorted(known)
165 # Layout circular determinista: posición decorativa, sin distancia comercial implícita.
166 positions = {
167 key: (math.cos(2 * math.pi * i / len(ordered)), math.sin(2 * math.pi * i / len(ordered)))
168 for i, key in enumerate(ordered)
169 }
170 if focus_id:
171 positions[focus_id] = (0, 0)
172 ranked = {row["company_id"]: row for row in report["companies"]}
173 focus = ranked.get(focus_id, {})
174 support_ids = {
175 identifier
176 for neighbor in focus.get("features", {})
177 .get("confirmed_neighbors", {})
178 .get("neighbors", [])
179 for identifier in neighbor["relationship_ids"]
180 }
181 nodes = []
182 for key in sorted(shown):
183 company = known[key]
184 incident = [
185 row
186 for row in view["relationships"]
187 if key in (row["from_company_id"], row["to_company_id"])
188 and row["relationship_type"] != "REFERS_TO"
189 ]
190 usable = any(
191 diagnostics[row["relationship_id"]]["scoring_status"] == "CONFIRMED" for row in incident
192 )
193 state = (
194 ("Con vínculo confirmado utilizable" if usable else "Sin vínculo confirmado utilizable")
195 if incident
196 else "Sin vínculo económico observado"
197 )
198 rank = ranked.get(key, {})
199 nodes.append(
200 {
201 "id": key,
202 "name": company["attributes"]["display_name"]["value"] or key,
203 "label": textwrap.fill(company["attributes"]["display_name"]["value"] or key, 18)
204 if len(shown) <= 10
205 else "",
206 "sector": sector(company),
207 "x": positions[key][0],
208 "y": positions[key][1],
209 "label_x": positions[key][0] * 1.22,
210 "label_y": positions[key][1] * 1.22 if key != focus_id else -0.22,
211 "size": 650 if key == focus_id else 180,
212 "scope": "Selección" if key in ids else "Vecino de contexto",
213 "state": state,
214 "identity": company["identity_status"],
215 "baseline_score": rank.get("baseline_score"),
216 "graph_score": rank.get("graph_score"),
217 }
218 )
219 edges = []
220 for row in claims:
221 left, right = row["from_company_id"], row["to_company_id"]
222 edges.append(
223 {
224 "id": row["relationship_id"],
225 "from": left,
226 "to": right,
227 "from_name": known[left]["attributes"]["display_name"]["value"] or left,
228 "to_name": known[right]["attributes"]["display_name"]["value"] or right,
229 "type": TYPES[row["relationship_type"]],
230 "status": STATES[row["verification_status"]],
231 "scoring_status": diagnostics[row["relationship_id"]]["scoring_status"],
232 "supports_focus": row["relationship_id"] in support_ids,
233 "emphasis": 4 if row["relationship_id"] in support_ids else 1.5,
234 "opacity": 1 if row["relationship_id"] in support_ids else 0.5,
235 "source": ", ".join(sorted({r["source_type"] for r in row["status_history"]})),
236 "evidence": ", ".join(r["observation_id"] for r in row["status_history"]),
237 "x": positions[left][0],
238 "y": positions[left][1],
239 "x2": positions[right][0],
240 "y2": positions[right][1],
241 }
242 )
243 return {"nodes": nodes, "edges": edges}

374def atlas_context(cut: str, ledger_path: Path, env_file: Path) -> dict:
375 """Lectura explícita, base dedicada y paridad; errores deliberadamente sin detalle remoto."""
376 try:
377 local = build_context(load_observations(ledger_path), cut)
378 uri, database = load_config(env_file)
379 if database != MVP_DATABASE:
380 raise ValueError("Base no autorizada")
381 with create_client(uri) as client:
382 rows = read_observations(client[database], cut)
383 context = build_context(rows, cut)
384 except (
385 ValueError,
386 ConnectionError,
387 OSError,
388 TypeError,
389 KeyError,
390 AttributeError,
391 PyMongoError,
392 ):
393 raise ValueError(
394 "No se pudo leer Atlas. Revisa configuración local, base dedicada, DNS, red, "
395 "TLS y permisos de lectura. No se cargaron cifras de respaldo."
396 ) from None
397 if context["ranking"]["visible_input_sha256"] != local["ranking"]["visible_input_sha256"]:
398 raise ValueError(
399 "Datos distintos del ledger aprobado. Demostración comparable bloqueada; "
400 "investiga la diferencia sin modificar Atlas."
401 )
402 return {"context": context, "read_at": datetime.now(UTC).isoformat()}
