39def parse_time(value: str | datetime) -> datetime:
40 """Aceptar un corte nombrado o un instante con zona; nunca suponer fin de día."""
41 try:
42 instant = (
43 value
44 if isinstance(value, datetime)
45 else datetime.fromisoformat(CUTS.get(value, value).replace("Z", "+00:00"))
46 )
47 if instant.tzinfo is None or instant.utcoffset() is None:
48 raise ValueError
49 return instant.astimezone(UTC)
50 except (ValueError, TypeError, AttributeError):
51 raise ValueError(
52 "La fecha debe ser T0/T1/T2 o un instante ISO 8601 con zona horaria."
53 ) from None

56def iso_time(value: str | datetime) -> str:
57 return parse_time(value).isoformat().replace("+00:00", "Z")

74def validate_observations(rows: list[dict]) -> None:
75 """Validar el ledger observable antes de consultarlo o enviarlo a MongoDB."""
76 seen = set()
77 for row in rows:
78 identifier = row.get("observation_id", "sin ID")
79 assert_no_private_fields(row)
80 if set(row) != OBSERVATION_FIELDS:
81 raise ValueError(f"{identifier}: campos observables incompletos o desconocidos.")
82 if not isinstance(identifier, str) or not identifier or identifier in seen:
83 raise ValueError(f"{identifier}: ID de observación vacío o duplicado.")
84 seen.add(identifier)
85 if row["source_type"] not in SOURCES or row["verification_status"] not in STATUSES:
86 raise ValueError(f"{identifier}: fuente o estado desconocido.")
87 for field in ("subject_id", "source_record_id", "evidence_note"):
88 if not isinstance(row[field], str) or not row[field].strip():
89 raise ValueError(f"{identifier}: falta {field}.")
90 dates = [parse_time(row[field]) for field in ("observed_at", "recorded_at", "available_at")]
91 if not dates[0] <= dates[1] <= dates[2]:
92 raise ValueError(f"{identifier}: fechas fuera de orden.")
93 kind, attribute, value = row["subject_type"], row["attribute"], row["value"]
94 if kind == "COMPANY":
95 if attribute not in COMPANY_ATTRIBUTES or not row["subject_id"].startswith("CO-"):
96 raise ValueError(f"{identifier}: atributo o identidad de empresa inválido.")
97 if attribute in {"display_name", "sector", "size"}:
98 valid = isinstance(value, str) and bool(value.strip())
99 elif attribute == "service_active":
100 valid = isinstance(value, bool)
101 elif attribute in {"branch_count", "review_count"}:
102 valid = type(value) is int and value >= (1 if attribute == "branch_count" else 0)
103 else:
104 valid = type(value) in (int, float) and 0 <= value <= 5
105 if not valid:
106 raise ValueError(f"{identifier}: valor incompatible con {attribute}.")
107 elif kind == "RELATIONSHIP":
108 required = {"from_company_id", "to_company_id", "relationship_type"}
109 if (
110 attribute != "relationship_claim"
111 or not isinstance(value, dict)
112 or set(value) != required
113 ):
114 raise ValueError(f"{identifier}: contrato de relación inválido.")
115 if value["relationship_type"] not in RELATION_TYPES:
116 raise ValueError(f"{identifier}: tipo de relación desconocido.")
117 left, right = value["from_company_id"], value["to_company_id"]
118 if left == right or not all(
119 isinstance(v, str) and v.startswith("CO-") for v in (left, right)
120 ):
121 raise ValueError(f"{identifier}: extremos de relación inválidos o auto-relación.")
122 if value["relationship_type"] == "PARTNERS_WITH" and left > right:
123 raise ValueError(f"{identifier}: alianza sin orientación canónica.")
124 if not row["subject_id"].startswith("LINK-"):
125 raise ValueError(f"{identifier}: identidad de relación inválida.")
126 elif kind == "SOURCE_RECORD":
127 if attribute != "alias_name" or not isinstance(value, str) or not value.strip():
128 raise ValueError(f"{identifier}: contrato de alias inválido.")
129 if not row["subject_id"].startswith("REC-"):
130 raise ValueError(f"{identifier}: identidad de registro inválida.")
131 else:
132 raise ValueError(f"{identifier}: tipo de sujeto desconocido.")
133
134 first_known = {}
135 for row in rows:
136 if row["subject_type"] == "COMPANY" and row["attribute"] == "display_name":
137 key = row["subject_id"]
138 first_known[key] = min(
139 first_known.get(key, parse_time(row["available_at"])),
140 parse_time(row["available_at"]),
141 )
142 for row in rows:
143 endpoints = []
144 if row["subject_type"] == "COMPANY":
145 endpoints = [row["subject_id"]]
146 elif row["subject_type"] == "RELATIONSHIP":
147 endpoints = [row["value"][key] for key in ("from_company_id", "to_company_id")]
148 if any(
149 key not in first_known or first_known[key] > parse_time(row["available_at"])
150 for key in endpoints
151 ):
152 raise ValueError(
153 f"{row['observation_id']}: empresa huérfana o no identificable al conocer el dato."
154 )
